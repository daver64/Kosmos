cmake_minimum_required(VERSION 3.16)
project(shipedit CXX)

# ShipEdit is a plain GTK 3 program (cairo draws the preview's overlay). It reads and writes ship.json files and does not talk to the running game.
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
# The default preview uses GtkGLArea; cairo is used if OpenGL is unavailable.
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
pkg_check_modules(GMODULE REQUIRED IMPORTED_TARGET gmodule-2.0)
find_path(KOSMOS_TOOL_VULKAN_INCLUDE vulkan/vulkan.h)
if(KOSMOS_TOOL_VULKAN_INCLUDE)
    set(KOSMOS_TOOL_VULKAN_DEFAULT ON)
else()
    set(KOSMOS_TOOL_VULKAN_DEFAULT OFF)
endif()
option(KOSMOS_TOOL_VULKAN "Build optional Vulkan rendering for GTK tools" ${KOSMOS_TOOL_VULKAN_DEFAULT})
if(KOSMOS_TOOL_VULKAN AND NOT KOSMOS_TOOL_VULKAN_INCLUDE)
    message(FATAL_ERROR "KOSMOS_TOOL_VULKAN requires Vulkan headers")
endif()

set(SHIPEDIT_JSON_INCLUDE "" CACHE PATH "Folder holding json.hpp (nlohmann)")
set(SHIPEDIT_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/bin" CACHE PATH "Where the shipedit program is written")

add_executable(shipedit src/main.cpp src/model.cpp src/mesh.cpp src/preview.cpp src/gl_preview.cpp ../tool_vulkan.cpp ../../src/part_kit.cpp)
target_compile_features(shipedit PRIVATE cxx_std_17)
target_include_directories(shipedit PRIVATE src ${SHIPEDIT_JSON_INCLUDE})
target_link_libraries(shipedit PRIVATE PkgConfig::GTK3 PkgConfig::EPOXY PkgConfig::GMODULE)
if(KOSMOS_TOOL_VULKAN)
    target_compile_definitions(shipedit PRIVATE KOSMOS_TOOL_VULKAN)
    target_include_directories(shipedit PRIVATE ${KOSMOS_TOOL_VULKAN_INCLUDE})
endif()
set_target_properties(shipedit PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${SHIPEDIT_OUTPUT_DIR})
if(COMMAND kosmos_expose_tool)
    kosmos_expose_tool(shipedit)
endif()
