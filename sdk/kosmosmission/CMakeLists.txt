cmake_minimum_required(VERSION 3.16)
project(kosmosmission CXX)

# KosmosMission is a GTK 3 mission planner with OpenGL and optional Vulkan views. The planning core (core/) has no GTK or GL in it:
# see tests/missioncore_test.cpp. The program talks to the game, optionally, through the SDK's client library.
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
pkg_check_modules(GMODULE REQUIRED IMPORTED_TARGET gmodule-2.0)
find_path(KOSMOS_TOOL_VULKAN_INCLUDE vulkan/vulkan.h)
if(KOSMOS_TOOL_VULKAN_INCLUDE)
    set(KOSMOS_TOOL_VULKAN_DEFAULT ON)
else()
    set(KOSMOS_TOOL_VULKAN_DEFAULT OFF)
endif()
option(KOSMOS_TOOL_VULKAN "Build optional Vulkan rendering for GTK tools" ${KOSMOS_TOOL_VULKAN_DEFAULT})
if(KOSMOS_TOOL_VULKAN AND NOT KOSMOS_TOOL_VULKAN_INCLUDE)
    message(FATAL_ERROR "KOSMOS_TOOL_VULKAN requires Vulkan headers")
endif()

set(KOSMOS_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Kosmos SDK folder")
if(NOT TARGET kosmos::link)
    add_subdirectory(${KOSMOS_SDK_DIR} kosmos-sdk EXCLUDE_FROM_ALL)
endif()

set(KOSMOSMISSION_JSON_INCLUDE "" CACHE PATH "Folder holding json.hpp (nlohmann)")
set(KOSMOSMISSION_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/bin" CACHE PATH "Where the kosmosmission program is written")

add_executable(kosmosmission src/main.cpp src/gl_view.cpp ../tool_vulkan.cpp core/orbit.cpp core/bodies.cpp core/transfer.cpp core/budget.cpp core/payload.cpp core/mission.cpp)
target_compile_features(kosmosmission PRIVATE cxx_std_17)
target_include_directories(kosmosmission PRIVATE src core ${KOSMOSMISSION_JSON_INCLUDE})
target_link_libraries(kosmosmission PRIVATE kosmos::link PkgConfig::GTK3 PkgConfig::EPOXY PkgConfig::GMODULE)
if(KOSMOS_TOOL_VULKAN)
    target_compile_definitions(kosmosmission PRIVATE KOSMOS_TOOL_VULKAN)
    target_include_directories(kosmosmission PRIVATE ${KOSMOS_TOOL_VULKAN_INCLUDE})
endif()
set_target_properties(kosmosmission PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${KOSMOSMISSION_OUTPUT_DIR})
if(COMMAND kosmos_expose_tool)
    kosmos_expose_tool(kosmosmission)
endif()
